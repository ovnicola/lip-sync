(()=>{

function fill(g,x,y,w,h,color){g.fillStyle=color;g.fillRect(x,y,w,h);}
function bars(g,w,h,level,y=0,height=h){const l=level;const colors=[[l,l,l],[l,l,0],[0,l,l],[0,l,0],[l,0,l],[l,0,0],[0,0,l]];colors.forEach((v,i)=>fill(g,Math.floor(i*w/7),y,Math.ceil(w/7)+1,height,'rgb('+v.join(',')+')'));}
function raster(g,w,h,type,p){
 const image=g.createImageData(w,h),d=image.data;
 const bands=[.015625,.03125,.0625,.125,.25,.45];
 const cx=(w-1)/2,cy=(h-1)/2,rmax=Math.max(1,Math.min(w,h)*.5);
 const cols=new Float64Array(w),rows=new Float64Array(h);
 if(type==='sweep')for(let x=0;x<w;x++)cols[x]=Math.cos(2*Math.PI*.45*x*x/(2*Math.max(1,w-1)));
 if(type==='zoneplate'){
  for(let x=0;x<w;x++)cols[x]=(x-cx)*(x-cx);
  for(let y=0;y<h;y++)rows[y]=(y-cy)*(y-cy);
 }
 for(let y=0;y<h;y++)for(let x=0;x<w;x++){
  let v=0;
  if(type==='multiburstV'){const b=Math.min(5,Math.floor(x*6/w));v=127.5+127.5*Math.cos(2*Math.PI*bands[b]*x);}
  else if(type==='multiburstH'){const b=Math.min(5,Math.floor(y*6/h));v=127.5+127.5*Math.cos(2*Math.PI*bands[b]*y);}
  else if(type==='zoneplate')v=127.5+127.5*Math.cos(Math.PI*.45*(cols[x]+rows[y])/rmax);
  else if(type==='sweep')v=127.5+127.5*cols[x];
  else if(type==='pixelV')v=Math.floor(x/p)%2?255:0;
  else if(type==='pixelH')v=Math.floor(y/p)%2?255:0;
  else if(type==='pixelCheck')v=(Math.floor(x/p)+Math.floor(y/p))%2?255:0;
  const i=(y*w+x)*4;d[i]=d[i+1]=d[i+2]=Math.round(v);d[i+3]=255;
 }
 g.putImageData(image,0,0);
}
function build(type,w,h){
 const c=document.createElement('canvas');c.width=w;c.height=h;const g=c.getContext('2d');fill(g,0,0,w,h,'#000');
 if(type==='bars75')bars(g,w,h,191);
 else if(type==='bars100')bars(g,w,h,255);
 else if(type==='smpteVisual'){
  bars(g,w,h,191,0,h*2/3);
  ['#0000bf','#131313','#bf00bf','#131313','#00bfbf','#131313','#bfbfbf'].forEach((color,i)=>fill(g,Math.floor(i*w/7),h*2/3,Math.ceil(w/7)+1,h/12+1,color));
  // Composition volontairement visuelle : pas de référence analogique I/Q ni de sous-noir.
  const patches=[[0,.18,'#003e64'],[.18,.18,'#fff'],[.36,.18,'#390076'],[.54,.18,'#131313'],[.72,.07,'#000'],[.79,.07,'#090909'],[.86,.07,'#131313'],[.93,.07,'#131313']];
  patches.forEach(([x,p,color])=>fill(g,w*x,h*.75,w*p+1,h*.25,color));
 }else if(['multiburstV','multiburstH','zoneplate','sweep','pixelV','pixelH','pixelCheck'].includes(type))raster(g,w,h,type,Number(pitch.value));
 else if(type==='graySteps'||type==='nearBlack'){
  const levels=type==='nearBlack'?[0,2,4,6,8,12,16,24,32]:Array.from({length:11},(_,i)=>Math.round(i*255/10));
  levels.forEach((v,i)=>fill(g,Math.floor(i*w/levels.length),0,Math.ceil(w/levels.length)+1,h,'rgb('+[v,v,v].join(',')+')'));
 }else if(type==='grayRamp'){
  for(let x=0;x<w;x++){const v=Math.round(x/Math.max(1,w-1)*255);fill(g,x,0,1,h,'rgb('+[v,v,v].join(',')+')');}
 }else if(type==='diagonals'){
  g.strokeStyle='#fff';g.lineWidth=1;
  for(const angle of [5,15,30,45,60,75,85]){g.save();g.translate(w/2,h/2);g.rotate(angle*Math.PI/180);g.beginPath();g.moveTo(-w,-h*.12);g.lineTo(w,-h*.12);g.moveTo(-w,h*.12);g.lineTo(w,h*.12);g.stroke();g.restore();}
 }else if(type==='geometry'){
  const rw=Math.min(w,h*16/9),rh=rw*9/16,ox=(w-rw)/2,oy=(h-rh)/2;
  fill(g,ox,oy,rw,rh,'#292929');g.lineWidth=Math.max(1,w/900);g.strokeStyle='#fff';
  for(const k of [1,.9,.8])g.strokeRect(ox+rw*(1-k)/2+1,oy+rh*(1-k)/2+1,rw*k-2,rh*k-2);
  g.beginPath();g.arc(w/2,h/2,rh*.38,0,Math.PI*2);g.moveTo(w/2,oy);g.lineTo(w/2,oy+rh);g.moveTo(ox,h/2);g.lineTo(ox+rw,h/2);g.stroke();
 }
 return c;
}
